## Usage

```tsx
import { ProList } from "@/components/pro/list"

<ProList
  data={projects}
  rowKey="id"
  variant="outline"
  direction="vertical"
  split
  search={{ placeholder: "Search projects..." }}
  filters={[
    {
      key: "status",
      placeholder: "Status",
      options: [
        { label: "Active", value: "active" },
        { label: "Paused", value: "paused" },
      ],
    },
  ]}
  renderItem={(project) => <ProjectItem project={project} />}
/>
```

Use `request` when pagination, search, and filters should be handled by your API.

```tsx
<ProList
  rowKey="id"
  request={(state) =>
    fetchProjects({
      page: state.pagination.pageIndex + 1,
      pageSize: state.pagination.pageSize,
      keyword: state.search,
      status: state.filters.status,
    })
  }
  renderItem={(project) => <ProjectItem project={project} />}
/>
```
