## Usage

```tsx
import { Copy, Maximize2, Minimize2, RefreshCw } from "lucide-react"
import { CopyButton, ProButton } from "@/components/pro/base/button"

<CopyButton
  icon={<Copy />}
  tooltip="Copy link"
  copy={() => window.location.href}
/>

<ProButton tooltip="Fullscreen" size="icon">
  <Maximize2 />
</ProButton>
```

`ProButton` owns the shadcn-compatible button variants, loading state, and optional tooltip composition without taking over action logic. `CopyButton` adds copy feedback. Refresh and fullscreen actions are regular `ProButton` instances composed with lucide icons at the call site. Copy success temporarily changes the icon to a green check; failed copy shows an error icon and tooltip.

`useFullscreen` lives in `base/hooks/use-fullscreen` and centralizes controlled/uncontrolled fullscreen behavior for components such as editors, viewers, previews, and panels.

## Examples

### Async Action

```tsx
<ProButton
  tooltip="Refresh"
  loading={isRefreshing}
  disabled={isRefreshing}
  onClick={() => refresh()}
>
  <RefreshCw />
  Refresh
</ProButton>
```

`ProButton` does not manage async click state. Pass `loading` explicitly when an action is pending.

### Copy

```tsx
<CopyButton
  icon={<Copy />}
  tooltip="Copy API key"
  copy={apiKey}
/>
```

### Fullscreen

```tsx
import { useFullscreen } from "@/components/pro/base/hooks/use-fullscreen"
import { ProButton } from "@/components/pro/base/button"

function PreviewPanel() {
  const { ref, fullscreen, setFullscreen } = useFullscreen({
    mode: "screen",
  })

  return (
    <div ref={ref}>
      <ProButton
        tooltip={fullscreen ? "Exit fullscreen" : "Fullscreen"}
        size="icon"
        onClick={() => setFullscreen(!fullscreen)}
      >
        {fullscreen ? <Minimize2 /> : <Maximize2 />}
      </ProButton>
      <div>{fullscreen ? "Fullscreen" : "Inline"}</div>
    </div>
  )
}
```

## API Reference

### ProButton

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `variant` | `'default' \| 'destructive' \| 'outline' \| 'secondary' \| 'ghost' \| 'link'` | `'default'` | Visual button style. |
| `size` | `'default' \| 'xs' \| 'sm' \| 'lg' \| 'icon' \| 'icon-xs' \| 'icon-sm' \| 'icon-lg'` | `'default'` | Button height and padding. Icon sizes are square. |
| `asChild` | `boolean` | `false` | Renders the shadcn slot child instead of a native button. |
| `loading` | `boolean` | `false` | Forces loading state. |
| `disabled` | `boolean` | `false` | Disables the button. Loading also disables it. |
| `onClick` | `(event) => void` | — | Native click handler. Async state is controlled by passing `loading`. |
| `type` | `'button' \| 'submit' \| 'reset'` | `'button'` | Native button type. Defaults to `button` to avoid accidental form submission. |
| `tooltip` | `ReactNode` | — | Optional tooltip content. If omitted, only the button is rendered. |
| `aria-label` | `string` | tooltip text | Accessible label. String tooltip content is used as a fallback. |
| `children` | `ReactNode` | — | Button label/content. |

Other button HTML props are passed to the rendered button or slot child.

### CopyButton

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `copy` | `string \| () => string \| Promise<string>` | — | Text or async text resolver to copy. |
| `icon` | `ReactNode` | — | Icon shown before the label while idle. Success and error states replace it. |
| `tooltip` | `ReactNode` | — | Default tooltip before feedback state is shown. |

`CopyButton` also accepts `ProButton` props.

### useFullscreen

| Option | Type | Default | Description |
| --- | --- | --- | --- |
| `fullscreen` | `boolean` | — | Controlled fullscreen state. |
| `defaultFullscreen` | `boolean` | `false` | Initial uncontrolled fullscreen state. |
| `onFullscreenChange` | `(fullscreen: boolean) => void` | — | Called when fullscreen state changes. |
| `mode` | `'fixed' \| 'screen'` | `'fixed'` | Fixed-position simulated fullscreen or browser Fullscreen API. |
| `ref` | `RefObject<HTMLDivElement \| null>` | internal ref | Optional target element ref. |

| --- | --- | --- |

| `fullscreen` | `boolean` | Current fullscreen state. |

import { type RefObject, useEffect, useRef, useState } from 'react'

export function useFullscreen({
  fullscreen: controlledFullscreen,
  defaultFullscreen = false,
  onFullscreenChange,
  mode = 'fixed',
  ref,
}: {
  fullscreen?: boolean
  defaultFullscreen?: boolean
  onFullscreenChange?: (fullscreen: boolean) => void
  mode?: 'fixed' | 'screen'
  ref?: RefObject<HTMLDivElement | null>
} = {}) {
  const internalRef = useRef<HTMLDivElement>(null)
  const targetRef = ref ?? internalRef
  const [uncontrolledFullscreen, setUncontrolledFullscreen] = useState(defaultFullscreen)
  const fullscreen = controlledFullscreen ?? uncontrolledFullscreen

  function setFullscreen(nextFullscreen: boolean) {
    if (controlledFullscreen === undefined) setUncontrolledFullscreen(nextFullscreen)
    onFullscreenChange?.(nextFullscreen)
  }
  useEffect(() => {
    if (mode !== 'screen') return

    function handleFullscreenChange() {
      if (document.fullscreenElement !== targetRef.current) {
        setFullscreen(false)
      }
    }

    document.addEventListener('fullscreenchange', handleFullscreenChange)
    return () => document.removeEventListener('fullscreenchange', handleFullscreenChange)
  }, [controlledFullscreen, mode, onFullscreenChange, targetRef])

  useEffect(() => {
    if (mode !== 'screen') return

    const element = targetRef.current
    if (!element) return

    if (fullscreen) {
      if (document.fullscreenElement !== element) {
        void element.requestFullscreen?.()?.catch(() => setFullscreen(false))
      }
      return
    }

    if (document.fullscreenElement === element) {
      void document.exitFullscreen?.()?.catch(() => setFullscreen(true))
    }
  }, [controlledFullscreen, fullscreen, mode, onFullscreenChange, targetRef])

  return {
    ref: targetRef,
    fullscreen,
    setFullscreen,
  }
}
